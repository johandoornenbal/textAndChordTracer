cmake_minimum_required(VERSION 3.22)
if(CMAKE_HOST_APPLE)
  set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Apple Silicon release architecture")
  if(NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "Release target is native Apple Silicon arm64")
  endif()
  set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0 CACHE STRING "Minimum macOS version")
endif()
project(TextAndChordTracer VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(TRACER_CORE_ONLY "Build musical analysis tests without GUI or speech dependencies" OFF)
add_library(tracer_analysis Source/MusicAnalysis.cpp)
set_target_properties(tracer_analysis PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(tracer_analysis PUBLIC Source)
if(BUILD_TESTING)
  add_executable(music_tests tests/MusicTests.cpp)
  target_link_libraries(music_tests PRIVATE tracer_analysis)
  add_test(NAME musical_analysis COMMAND music_tests)
endif()
if(TRACER_CORE_ONLY)
  return()
endif()
include(FetchContent)
set(JUCE_PATH "" CACHE PATH "Existing JUCE 8.0.12 checkout")
if(JUCE_PATH)
  add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE")
else()
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
  FetchContent_MakeAvailable(JUCE)
endif()
option(TRACER_WITH_WHISPER "Enable local speech recognition (required for release builds)" ON)
if(TRACER_WITH_WHISPER)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependencies" FORCE)
  set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(GGML_NATIVE OFF CACHE BOOL "Portable backend" FORCE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "Embed Metal kernels" FORCE)
  set(WHISPER_PATH "" CACHE PATH "Existing whisper.cpp v1.7.6 checkout")
  if(WHISPER_PATH)
    add_subdirectory("${WHISPER_PATH}" "${CMAKE_BINARY_DIR}/whisper")
  else()
    FetchContent_Declare(whisper GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG a8d002cfd879315632a579e73f0148d06959de36)
    FetchContent_MakeAvailable(whisper)
  endif()
endif()
juce_add_plugin(TextAndChordTracer
  COMPANY_NAME "Doornenbal Audio" BUNDLE_ID "nl.doornenbal.textandchordtracer"
  PLUGIN_MANUFACTURER_CODE Jdau PLUGIN_CODE Tct1 PRODUCT_NAME "Text and Chord Tracer"
  FORMATS VST3 IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
  EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Tools)
juce_generate_juce_header(TextAndChordTracer)
target_sources(TextAndChordTracer PRIVATE Source/Plugin.cpp Source/TraceWorker.cpp Source/SpeechEngine.cpp)
target_compile_definitions(TextAndChordTracer PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 TRACER_WITH_WHISPER=$<BOOL:${TRACER_WITH_WHISPER}>)
target_link_libraries(TextAndChordTracer PRIVATE tracer_analysis juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
if(TRACER_WITH_WHISPER)
  target_link_libraries(TextAndChordTracer PRIVATE whisper)
endif()
add_custom_command(TARGET TextAndChordTracer_VST3 POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/third_party"
    "${CMAKE_BINARY_DIR}/TextAndChordTracer_artefacts/$<CONFIG>/VST3/Text and Chord Tracer.vst3/Contents/Resources/Licenses")
if(BUILD_TESTING)
  add_executable(plugin_checks tests/PluginChecks.cpp)
  target_include_directories(plugin_checks PRIVATE "${JUCE_MODULES_DIR}")
  target_compile_definitions(plugin_checks PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  target_link_libraries(plugin_checks PRIVATE TextAndChordTracer)
  add_test(NAME plugin_state_export_ui COMMAND plugin_checks)
  add_executable(worker_checks tests/WorkerChecks.cpp)
  target_include_directories(worker_checks PRIVATE "${JUCE_MODULES_DIR}")
  target_compile_definitions(worker_checks PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  target_link_libraries(worker_checks PRIVATE TextAndChordTracer)
  add_test(NAME live_analysis_worker COMMAND worker_checks)
  if(TRACER_WITH_WHISPER)
    add_executable(speech_checks tests/SpeechChecks.cpp)
    target_include_directories(speech_checks PRIVATE "${JUCE_MODULES_DIR}")
    target_compile_definitions(speech_checks PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
    target_link_libraries(speech_checks PRIVATE TextAndChordTracer)
    if(WHISPER_PATH)
      set(TRACER_SPEECH_FIXTURE "${WHISPER_PATH}/samples/jfk.wav")
    else()
      set(TRACER_SPEECH_FIXTURE "${whisper_SOURCE_DIR}/samples/jfk.wav")
    endif()
    add_test(NAME local_speech_backend COMMAND speech_checks "${TRACER_SPEECH_FIXTURE}")
    set_tests_properties(local_speech_backend PROPERTIES SKIP_RETURN_CODE 77)
  endif()
endif()
